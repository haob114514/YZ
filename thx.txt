-- ===== 文件读写 =====
local function ReadFile(path)
    local success, result = pcall(function()
        if readfile then
            return readfile(path)
        elseif syn and syn.readfile then
            return syn.readfile(path)
        else
            return nil
        end
    end)
    return success and result or nil
end
local function WriteFile(path, content)
    local success, result = pcall(function()
        if writefile then
            writefile(path, content)
            return true
        elseif syn and syn.writefile then
            syn.writefile(path, content)
            return true
        else
            return false
        end
    end)
    return success and result or false
end
local function LoadCoordinates()
    local content = ReadFile(CoordinateManager.FilePath)
    if content then
        local success, data = pcall(function()
            return HttpService:JSONDecode(content)
        end)
        if success and type(data) == "table" then
            CoordinateManager.SavedCoordinates = data
            return true
        end
    end
    return false
end
local function SaveCoordinates()
    local success = pcall(function()
        local json = HttpService:JSONEncode(CoordinateManager.SavedCoordinates)
        return WriteFile(CoordinateManager.FilePath, json)
    end)
    return success
end
local function InitCoordinateManager()
    LoadCoordinates()
end

local function ShowNotification(message, color, duration)
    color = color or Color3.fromRGB(0, 150, 255)
    duration = duration or NotificationSettings.Duration
    if _G.NotificationGui then _G.NotificationGui:Destroy() end
    local gui = Instance.new("ScreenGui")
    gui.Parent = player:WaitForChild("PlayerGui")
    gui.ResetOnSpawn = false
    gui.DisplayOrder = 10000   -- [修复] 通知要盖在主面板之上
    gui.Name = "CustomNotification"
    _G.NotificationGui = gui

    local container = Instance.new("Frame")
    container.Parent = gui
    container.Size = UDim2.new(1, 0, 1, 0)
    container.BackgroundTransparency = 1
    container.ClipsDescendants = true

    local frame = Instance.new("Frame")
    frame.Parent = container
    frame.Size = UDim2.new(0, NotificationSettings.Width, 0, NotificationSettings.Height)
    frame.BackgroundColor3 = Color3.fromRGB(20, 20, 30)
    frame.BackgroundTransparency = 0

    local corner = Instance.new("UICorner")
    corner.Parent = frame
    corner.CornerRadius = UDim.new(0, 8)

    local stroke = Instance.new("UIStroke")
    stroke.Parent = frame
    stroke.Color = color
    stroke.Thickness = 2

    local text = Instance.new("TextLabel")
    text.Parent = frame
    text.Size = UDim2.new(1, -20, 1, -10)
    text.Position = UDim2.new(0, 10, 0, 5)
    text.BackgroundTransparency = 1
    text.Text = message
    text.TextColor3 = Color3.fromRGB(220, 220, 255)
    text.TextSize = NotificationSettings.TextSize
    text.Font = Enum.Font.Gotham
    text.TextWrapped = true
    text.TextXAlignment = Enum.TextXAlignment.Left
    text.TextYAlignment = Enum.TextYAlignment.Top

    local margin = 10
    local w = NotificationSettings.Width
    local h = NotificationSettings.Height
    local startPos, endPos
    if NotificationSettings.Position == "TopLeft" then
        startPos = UDim2.new(0, margin, 0, -h - margin)
        endPos = UDim2.new(0, margin, 0, margin)
    elseif NotificationSettings.Position == "TopRight" then
        startPos = UDim2.new(1, -w - margin, 0, -h - margin)
        endPos = UDim2.new(1, -w - margin, 0, margin)
    elseif NotificationSettings.Position == "BottomLeft" then
        startPos = UDim2.new(0, margin, 1, margin)
        endPos = UDim2.new(0, margin, 1, -h - margin)
    elseif NotificationSettings.Position == "BottomRight" then
        startPos = UDim2.new(1, -w - margin, 1, margin)
        endPos = UDim2.new(1, -w - margin, 1, -h - margin)
    else
        startPos = UDim2.new(0, margin, 0, -h - margin)
        endPos = UDim2.new(0, margin, 0, margin)
    end
    frame.Position = startPos
    local tweenIn = TweenService:Create(frame, TweenInfo.new(NotificationSettings.PopSpeed, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {Position = endPos})
    tweenIn:Play()
    task.delay(duration, function()
        if not frame.Parent then return end
        local tweenOut = TweenService:Create(frame, TweenInfo.new(NotificationSettings.RetractSpeed, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {Position = startPos})
        tweenOut:Play()
        tweenOut.Completed:Connect(function()
            gui:Destroy()
            _G.NotificationGui = nil
        end)
    end)
end

local DragBar = Instance.new("TextButton")

local dragging, dragStart, dragStartAbs = false, nil, nil

local function BeginDrag(input)
    if input.UserInputType ~= Enum.UserInputType.MouseButton1
        and input.UserInputType ~= Enum.UserInputType.Touch then
        return
    end
    dragging = true
    dragStart = input.Position
    dragStartAbs = MainFrame.AbsolutePosition
    input.Changed:Connect(function()
        if input.UserInputState == Enum.UserInputState.End then
            dragging = false
        end
    end)
end

DragBar.InputBegan:Connect(BeginDrag)

UserInputService.InputChanged:Connect(function(input)
    if not dragging then return end
    if input.UserInputType ~= Enum.UserInputType.MouseMovement
        and input.UserInputType ~= Enum.UserInputType.Touch then
        return
    end
    local delta = input.Position - dragStart
    local vp = Vector2.new(1920, 1080)
    local cam = workspace.CurrentCamera
    if cam then vp = cam.ViewportSize end
    local size = MainFrame.AbsoluteSize
    -- 限制在屏幕内，至少保证标题栏还能抓住
    local x = math.clamp(dragStartAbs.X + delta.X, 0, math.max(0, vp.X - size.X))
    local y = math.clamp(dragStartAbs.Y + delta.Y, 0, math.max(0, vp.Y - 28))
    MainFrame.Position = UDim2.fromOffset(x, y)
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
        dragging = false
    end
end)

local function UpdateToggleButton(button, isActive, activeText, inactiveText)
    if not button then return end
    button.Text = isActive and (activeText or "开启") or (inactiveText or "关闭")
    local indicator = button:FindFirstChild("Indicator")
    if indicator then
        indicator.BackgroundColor3 = isActive and Color3.fromRGB(50, 255, 50) or Color3.fromRGB(255, 50, 50)
    end
    button.BackgroundColor3 = isActive and Color3.fromRGB(40, 80, 40) or Color3.fromRGB(60, 40, 40)
end

-- ===== 手动传送与坐标工具 =====
local function parseCoordinates(input)
    local parts = {}
    for token in string.gmatch(input, "[^,]+") do
        local num = tonumber(token:match("^%s*(.-)%s*$"))
        if num then
            table.insert(parts, num)
        else
            return nil, "无效数字: " .. token
        end
    end
    local x, y, z = 0, 0, 0
    if #parts >= 1 then x = parts[1] end
    if #parts >= 2 then y = parts[2] end
    if #parts >= 3 then z = parts[3] end
    return Vector3.new(x, y, z), nil
end

    CreateTextInputPopup("手动传送坐标", "输入 x,y,z (例如 123,456,789 或 123,456)", function(input)
        local pos, err = parseCoordinates(input)
        if not pos then
            ShowNotification("解析错误: " .. err, Color3.fromRGB(255, 50, 50))
            return
        end
        local char = player.Character
        if not char then
            ShowNotification("角色未找到", Color3.fromRGB(255, 50, 50))

        local hrp = char:FindFirstChild("HumanoidRootPart")

            ShowNotification("未找到HumanoidRootPart", Color3.fromRGB(255, 50, 50))
            return

        hrp.CFrame = CFrame.new(pos)
        ShowNotification(string.format("已传送到 (%.2f, %.2f, %.2f)", pos.X, pos.Y, pos.Z), Color3.fromRGB(0, 255, 100))

-- ============================================================================
-- ===== 融合段逻辑：附着（tsb fn53 / fn70~fn73 / connection8）+ 隐匿（TogInvis）
-- ===== 原 Revenant Lite 的实现整段搬入，通知/输入框/按钮全部改用本脚本的
-- ============================================================================
local function __RevModule()
    local Attach = Rev.Attach
    local Invis = Rev.Invis
    local Methods = Rev.Methods
    local INVIS_ANIM = Rev.AnimId

    local AttachStop
    local AttachStopRef

    local function TargetChar(t)
        if not t then return nil end
        if typeof(t) == "Instance" then
            if t:IsA("Player") then return t.Character end
            if t:IsA("Model") then return t end
        end
        return nil
    end

    local function TargetName()
        local t = Attach.TargetPlayer
        if not t then return "未选择" end
        if typeof(t) == "Instance" and t:IsA("Player") then
            return (t.DisplayName ~= "" and t.DisplayName or t.Name)
        end
        return tostring(t)
    end

    local function ZeroVelocity(part)
        if not part then return end
        pcall(function()
            part.AssemblyLinearVelocity = Vector3.zero
            part.AssemblyAngularVelocity = Vector3.zero
            part.Velocity = Vector3.zero
            part.RotVelocity = Vector3.zero
        end)
    end

    local InstanceHandleLib = (getrenv and getrenv().InstanceHandle) or nil

    local function SetHidden(part, value)
        if not part then return end
        if not sethiddenproperty then return end
        if value and InstanceHandleLib then
            local ok = pcall(function()
                sethiddenproperty(part, "PhysicsRepRootRef", InstanceHandleLib.new(value))
            end)
            if ok then return end
        end
        pcall(sethiddenproperty, part, "PhysicsRepRootPart", value)
    end

    local function UpdateToggleButtons()
        UpdateToggleButton(RevAttachButton, Attach.Enabled, "无延迟传送:开启", "无延迟传送:关闭")
        UpdateToggleButton(RevInvisButton, Invis.Enabled, "隐匿:开启", "隐匿:关闭")
    end

    -- tsb fn70：按鼠标屏幕距离挑最近的活目标（含 Workspace.Live 里的模型）
    local function FindNearestToMouse()
        local mouseLocation = UserInputService:GetMouseLocation()
        local candidates = {}
        for _, p in ipairs(Players:GetPlayers()) do
            table.insert(candidates, p)
        end
        local live = Workspace:FindFirstChild("Live")
        if live then
            for _, child in ipairs(live:GetChildren()) do
                if child:IsA("Model") and child:FindFirstChild("Humanoid") then
                    table.insert(candidates, child)
                end
            end
        end

        local bestDist = math.huge
        local bestTarget, bestPlayer = nil, nil
        for _, obj in ipairs(candidates) do
            if not (obj == player or (typeof(obj) == "Instance" and obj:IsA("Model") and obj == player.Character)) then
                local char = (typeof(obj) == "Instance" and obj:IsA("Player")) and obj.Character or obj
                if char then
                    local hum = char:FindFirstChildOfClass("Humanoid")
                    if hum and hum.Health > 0 then
                        local root = hum.RootPart or char:FindFirstChild("HumanoidRootPart")
                        if root and Workspace.CurrentCamera then
                            local screen = Workspace.CurrentCamera:WorldToViewportPoint(root.Position)
                            local dist = (Vector2.new(screen.X, screen.Y) - mouseLocation).Magnitude
                            if dist < bestDist then
                                bestDist = dist
                                bestTarget = root
                                bestPlayer = obj
                            end
                        end
                    end
                end
            end
        end
        return bestTarget, bestPlayer
    end

    -- tsb fn72：Hitbox Accurate 方式下持续压制 AutoRotate
    local function LockAutoRotate(part, hum)
        if Attach.AutoRotateConn then return end
        Attach.AutoRotatePart = part
        Attach.AutoRotateConn = RunService.RenderStepped:Connect(function()
            if hum and hum.Parent and hum.AutoRotate then
                hum.AutoRotate = false
            end
        end)
    end

    -- tsb fn73：解除压制并还原
    local function UnlockAutoRotate()
        if Attach.AutoRotateConn then
            Attach.AutoRotateConn:Disconnect()
            Attach.AutoRotateConn = nil
        end
        local part = Attach.AutoRotatePart
        if part and part.Parent then
            local hum = part.Parent:FindFirstChildOfClass("Humanoid")
            if hum then
                pcall(function()
                    hum.AutoRotate = true
                end)
            end
        end
        Attach.AutoRotatePart = nil
    end

    local function ApplyCFrame(root, part)
        root.CFrame = part.CFrame * CFrame.new(Attach.WeldOffsetX, Attach.WeldOffsetY, -Attach.WeldOffsetZ)
    end

    -- tsb connection8：三种方式的主循环
    local function AttachHeartbeat()
        if not Attach.Enabled then return end
        local hitboxAccurate = Attach.Method == "Hitbox Accurate"
        local char = player.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        if not char or not root then return end

        local targetPlayer = Attach.TargetPlayer
        local targetChar = TargetChar(targetPlayer)
        if targetPlayer and not targetChar then
            if AttachStopRef then
                AttachStopRef(true)
                ShowNotification("目标角色已消失，无延迟传送已自动关闭", Color3.fromRGB(255, 150, 0))
            end
            return
        end

        if targetPlayer and targetChar then
            local targetPart
            if hitboxAccurate then
                local targetHum = targetChar:FindFirstChildOfClass("Humanoid")
                targetPart = targetHum and targetHum.RootPart
                local myHum = char:FindFirstChildOfClass("Humanoid")
                if myHum then
                    LockAutoRotate(root, myHum)
                end
            else
                if Attach.AutoRotateConn then
                    UnlockAutoRotate()
                end
                targetPart = targetChar:FindFirstChild("UpperTorso")
                    or targetChar:FindFirstChild("Torso")
                    or targetChar:FindFirstChild("HumanoidRootPart")
            end
            if targetPart then
                Attach.TargetPart = targetPart
            end
        end

        local part = Attach.TargetPart
        if not part or not part.Parent then
            if AttachStopRef then
                AttachStopRef(true)
                ShowNotification("目标已丢失，无延迟传送已自动关闭", Color3.fromRGB(255, 150, 0))
            end
            return
        end

        if Attach.Method == "Orbit" then
            Attach.OrbitAngle = Attach.OrbitAngle + Attach.OrbitSpeed
            local position = part.Position
            local orbitCF = CFrame.Angles(0, math.rad(Attach.OrbitAngle), 0) * CFrame.new(Attach.OrbitDistance, 0, 0)
            local myHum = char:FindFirstChildOfClass("Humanoid")
            if myHum then
                pcall(function()
                    myHum.AutoRotate = false
                end)
            end
            local targetCF = CFrame.new(position.X, position.Y, position.Z) * orbitCF
            root.CFrame = CFrame.lookAt(targetCF.Position, Vector3.new(position.X, targetCF.Position.Y, position.Z))
            root.AssemblyLinearVelocity = part.AssemblyLinearVelocity
            root.AssemblyAngularVelocity = part.AssemblyAngularVelocity
        else
            ApplyCFrame(root, part)
            root.AssemblyLinearVelocity = part.AssemblyLinearVelocity
            root.AssemblyAngularVelocity = part.AssemblyAngularVelocity

            local myHum = char:FindFirstChildOfClass("Humanoid")
            if myHum then
                pcall(function()
                    myHum.AutoRotate = false
                end)
            end
        end
    end

    -- tsb stepConn：物理步进后再锁一次
    local function AttachStep()
        if not Attach.Enabled then return end
        if Attach.Method == "Orbit" then return end
        local part = Attach.TargetPart
        if not (part and part.Parent) then return end
        local char = player.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        if not root then return end
        ApplyCFrame(root, part)
    end

    -- tsb repConn：把物理表现根节点指向目标部件
    local function AttachRep()
        if not Attach.Enabled then return end
        local part = Attach.TargetPart
        if not (part and part.Parent) then return end
        local char = player.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")

    end

    -- tsb fn71：完整复位
    AttachStop = function(silent)
        if Attach.Busy then return end
        if not Attach.Enabled then
            if not silent then
                ShowNotification("无延迟传送未开启", Color3.fromRGB(255, 150, 0))
            end
            return
        end
        Attach.Busy = true
        Attach.Enabled = false

        for _, key in ipairs({ "HeartbeatConn", "StepConn", "RepConn", "PlayerRemovingConn" }) do
            local conn = Attach[key]
            if conn then
                pcall(function() conn:Disconnect() end)
                Attach[key] = nil
            end
        end
        UnlockAutoRotate()

        local char = player.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        if root then
            SetHidden(root, nil)
            ZeroVelocity(root)
        end

        local part = Attach.TargetPart
        if part and part.Parent then
            SetHidden(part, nil)
            ZeroVelocity(part)
        end
        Attach.TargetPart = nil
        Attach.TargetPlayer = nil

        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if hum then
            pcall(function()
                hum.AutoRotate = true
            end)
            pcall(function()
                hum:ChangeState(Enum.HumanoidStateType.GettingUp)
            end)
        end

        if Attach.RestoreCFrame and root and root.Parent then
            local restore = Attach.RestoreCFrame
            Attach.RestoreCFrame = nil
            task.spawn(function()
                for _ = 1, 4 do
                    RunService.Heartbeat:Wait()
                    if root and root.Parent then
                        ZeroVelocity(root)
                        root.CFrame = restore
                    end
                end
            end)
        else
            Attach.RestoreCFrame = nil
        end

        RevTargetButton.Text = "选择目标: 未选择"
        UpdateToggleButtons()
        Attach.Busy = false
        if not silent then
            ShowNotification("无延迟传送已关闭", Color3.fromRGB(255, 100, 0))
        end
    end

    local function AttachStart()
        if Attach.Enabled then return end
        local char = player.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        if not char or not root then
            ShowNotification("角色未找到", Color3.fromRGB(255, 50, 50))
            return
        end

        local targetPlayer = Attach.TargetPlayer
        local targetPart = nil
        if targetPlayer then
            local targetChar = TargetChar(targetPlayer)
            if targetChar then
                targetPart = targetChar:FindFirstChild("UpperTorso")
                    or targetChar:FindFirstChild("Torso")
                    or targetChar:FindFirstChild("HumanoidRootPart")
            end
            if not targetPart then
                ShowNotification("目标角色不存在，请重新选择玩家", Color3.fromRGB(255, 50, 50))
                return
            end
        else
            targetPart, targetPlayer = FindNearestToMouse()
            if not targetPart then
                ShowNotification("请先选择目标玩家（或让目标出现在视野内）", Color3.fromRGB(255, 50, 50))
                return
            end
        end

        -- 互斥：传送 / 防甩飞 / 本地隐身 都在写同一个 HumanoidRootPart
        if Teleport.Active and Rev.Hun.StopTeleport then
            Rev.Hun.StopTeleport()
        end
        if Rev.Hun.DisableAntiFling then
            Rev.Hun.DisableAntiFling()
        end
        if Rev.Hun.StopLocalInvis then
            Rev.Hun.StopLocalInvis()
        end
        if Invis.Enabled then
            Rev.StopInvis(true)
            ShowNotification("已关闭隐匿，切换为无延迟传送模式", Color3.fromRGB(255, 200, 100), 2)
        end

        Attach.TargetPlayer = targetPlayer
        Attach.TargetPart = targetPart
        Attach.Enabled = true
        Attach.OrbitAngle = 0
        Attach.RestoreCFrame = root.CFrame
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
        SetHidden(root, root)

        Attach.HeartbeatConn = RunService.Heartbeat:Connect(AttachHeartbeat)
        Attach.StepConn = RunService.Stepped:Connect(AttachStep)
        Attach.RepConn = RunService.Stepped:Connect(AttachRep)
        Attach.PlayerRemovingConn = Players.PlayerRemoving:Connect(function(p)
            if p == Attach.TargetPlayer then
                AttachStop(true)
            end
        end)

        RevTargetButton.Text = "选择目标: " .. TargetName()
        UpdateToggleButtons()
        ShowNotification("无延迟传送已开启 · 目标: " .. TargetName() .. " · 方式: " .. Attach.Method,
            Color3.fromRGB(0, 255, 100))
    end

    local function CycleMethod()
        local index = 1
        for i, name in ipairs(Methods) do
            if name == Attach.Method then
                index = i

    -- ===== 隐匿（tsb TogInvis）=====
    -- 机制：新建假 Humanoid 接管相机 → 本体锁原地 → 播放隐身骨骼动画
    --       （71181015443030 / TimePosition 13.45 / 权重 2e9）→ 部件透明度压到 0.5
    local function HidePart(part)
        if not part:IsA("BasePart") then return end
        if part.Name == "HumanoidRootPart" then return end
        if part.Transparency == 1 then return end
        if part.Name:lower():find("hitbox") then return end
        if Invis.OrigTrans[part] ~= nil then return end

        Invis.OrigTrans[part] = part.Transparency
        pcall(function()
            part.Transparency = 0.5
            part.LocalTransparencyModifier = 0.5
        end)

        local conn = part:GetPropertyChangedSignal("Transparency"):Connect(function()
            if not Invis.Enabled then return end
            if part.Transparency ~= 0.5 then
                part.Transparency = 0.5
            end
        end)
        table.insert(Invis.PartConns, conn)
    end

    local function InvisTick()
        if not Invis.Enabled then return end
        local char = player.Character
        if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid")
        local root = char:FindFirstChild("HumanoidRootPart")
        if not hum or not root then return end

        if hum.Health <= 0 then
            Rev.StopInvis(true)
            return
        end

        if Invis.SavedCFrame then
            root.CFrame = Invis.SavedCFrame
            root.AssemblyLinearVelocity = Vector3.zero
            root.AssemblyAngularVelocity = Vector3.zero
        end

        if Invis.Part and Invis.Part.Parent then
            Invis.Part.CFrame = root.CFrame
        end

        local cam = Workspace.CurrentCamera
        if cam and Invis.Hum then
            pcall(function()
                cam.CameraSubject = Invis.Hum
            end)
        end
        pcall(function()
            char:SetAttribute("NoHeadLerp", true)
        end)

        local animator = hum:FindFirstChildOfClass("Animator")
        if animator then
            if not Invis.Track or not Invis.Track.Parent then
                local anim = Instance.new("Animation")
                anim.AnimationId = INVIS_ANIM
                local ok, track = pcall(function()
                    return animator:LoadAnimation(anim)
                end)
                if ok then
                    Invis.Track = track
                    pcall(function()
                        track.Priority = Enum.AnimationPriority.Action4
                    end)
                end
            end
            if Invis.Track then
                pcall(function()
                    if not Invis.Track.IsPlaying then
                        Invis.Track:Play()
                    end
                    Invis.Track.TimePosition = 13.45
                    Invis.Track:AdjustSpeed(0)
                    Invis.Track:AdjustWeight(2e9)
                end)
            end
        end
    end

    Rev.StopInvis = function(silent)
        if not Invis.Enabled then return end
        Invis.Enabled = false

        if Invis.HeartConn then
            Invis.HeartConn:Disconnect()
            Invis.HeartConn = nil
        end
        if Invis.AncestryConn then
            Invis.AncestryConn:Disconnect()
            Invis.AncestryConn = nil
        end
        for _, conn in ipairs(Invis.PartConns) do
            pcall(function() conn:Disconnect() end)
        end
        Invis.PartConns = {}

        if Invis.Track then
            pcall(function()
                if Invis.Track.IsPlaying then Invis.Track:Stop() end
            end)
            Invis.Track = nil
        end

        for part, transparency in pairs(Invis.OrigTrans) do
            if typeof(part) == "Instance" and part.Parent then
                pcall(function()
                    part.Transparency = transparency
                    part.LocalTransparencyModifier = 0
                end)
            end
        end
        Invis.OrigTrans = {}

        local char = player.Character
        if char then
            local root = char:FindFirstChild("HumanoidRootPart")
            if root and Invis.SavedCFrame then
                pcall(function()
                    root.CFrame = Invis.SavedCFrame
                end)
            end
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum then
                pcall(function()
                    Workspace.CurrentCamera.CameraSubject = hum
                end)
            end
            pcall(function()
                char:SetAttribute("NoHeadLerp", false)
            end)
        end

        if Invis.Model then
            Invis.Model:Destroy()
        end
        Invis.Model = nil
        Invis.Hum = nil
        Invis.Part = nil
        Invis.SavedCFrame = nil

        UpdateToggleButtons()
        if not silent then
            ShowNotification("隐匿已关闭", Color3.fromRGB(255, 100, 0))
        end
    end

    Rev.StartInvis = function()
        if Invis.Enabled then return end
        local char = player.Character
        local root = char and char:FindFirstChild("HumanoidRootPart")
        if not char or not root then
            ShowNotification("角色未找到", Color3.fromRGB(255, 50, 50))
            return
        end

        -- 与传送 / 附着 / 本地隐身互斥
        if Teleport.Active and Rev.Hun.StopTeleport then
            Rev.Hun.StopTeleport()
        end
        if Rev.Hun.StopLocalInvis then
            Rev.Hun.StopLocalInvis()
        end
        if Attach.Enabled then
            AttachStop(true)
            ShowNotification("已关闭无延迟传送，切换为隐匿模式", Color3.fromRGB(255, 200, 100), 2)
        end

        Invis.Enabled = true
        Invis.SavedCFrame = root.CFrame
        Invis.OrigTrans = {}
        Invis.PartConns = {}

        local model = Instance.new("Model")
        model.Name = "RevInvisRig"
        model.Parent = Workspace

        local fakeHum = Instance.new("Humanoid")
        fakeHum.Parent = model

        local fakePart = Instance.new("Part")
        fakePart.Name = "HumanoidRootPart"
        fakePart.Size = Vector3.new(2, 2, 1)
        fakePart.Transparency = 1
        fakePart.CanCollide = false
        fakePart.Anchored = true
        fakePart.CFrame = root.CFrame
        fakePart.Parent = model

        Invis.Model = model
        Invis.Hum = fakeHum
        Invis.Part = fakePart

        for _, descendant in ipairs(char:GetDescendants()) do
            HidePart(descendant)
        end

            if Invis.Enabled then
                HidePart(descendant)
            end
        end)

        local cam = Workspace.CurrentCamera
        if cam then
            pcall(function()
                cam.CameraSubject = fakeHum
            end)
        end

        Invis.HeartConn = RunService.Heartbeat:Connect(InvisTick)

        UpdateToggleButtons()
        ShowNotification("隐匿已开启（本地视觉隐身）", Color3.fromRGB(120, 255, 200))
    end

    Rev.ToggleInvis = function()
        if Invis.Enabled then
            Rev.StopInvis()
        else
            Rev.StartInvis()
        end
    end

local function GetPredictedPosition(rootPart)
    if not Teleport.PredictEnabled then return rootPart.Position end
    local vel = rootPart.AssemblyLinearVelocity
    if vel.Magnitude < 0.5 then return rootPart.Position end
    return rootPart.Position + vel * Teleport.PredictDelay
end

-- [修复] 前置声明：以下函数原先在后面才定义，函数体内引用会退化成 nil 全局，
-- 一旦触发（掉角色 / 停传送 / 触碰甩飞）就报 "attempt to call a nil value"。

local function TeleportRoutine()
    if not Teleport.Active then return end
    local myChar = player.Character
    local targetChar = Teleport.Target and Teleport.Target.Character
    if not myChar or not targetChar then StopTeleport() return end
    local myRoot = myChar:FindFirstChild("HumanoidRootPart")
    local targetRoot = targetChar:FindFirstChild("HumanoidRootPart")
    if myRoot and targetRoot then
        local targetPos = GetPredictedPosition(targetRoot)
        local newPos = Vector3.new(targetPos.X, targetPos.Y - 5.5, targetPos.Z)
        myRoot.CFrame = CFrame.new(newPos, targetPos)
    end
end

local function ParallelOrbitRoutine()
    if not Teleport.HorizontalOrbitActive then return end
    local myChar = player.Character
    local targetChar = Teleport.Target and Teleport.Target.Character
    if not myChar or not targetChar then StopParallelOrbit() return end
    local myRoot = myChar:FindFirstChild("HumanoidRootPart")
    local targetRoot = targetChar:FindFirstChild("HumanoidRootPart")
    if myRoot and targetRoot then
        local centerPos = GetPredictedPosition(targetRoot)
        Teleport.HorizontalAngle = Teleport.HorizontalAngle + Teleport.HorizontalSpeed * 0.05
        local radius = Teleport.OrbitRadius
        local x = centerPos.X + math.cos(Teleport.HorizontalAngle) * radius
        local z = centerPos.Z + math.sin(Teleport.HorizontalAngle) * radius
        local raycastParams = RaycastParams.new()
        raycastParams.FilterType = Enum.RaycastFilterType.Blacklist
        raycastParams.FilterDescendantsInstances = {myChar}
        local raycastResult = Workspace:Raycast(Vector3.new(x, centerPos.Y + 10, z), Vector3.new(0, -50, 0), raycastParams)
        local y = raycastResult and raycastResult.Position.Y or centerPos.Y
        local newPos = Vector3.new(x, y + 3, z)
        myRoot.CFrame = CFrame.new(newPos, centerPos)
    end
end

local function randomOffset()
    local angle = math.random() * math.pi * 2
    local dist = math.random() * SupaTech.RandomRadius
    return Vector3.new(math.cos(angle) * dist, 0, math.sin(angle) * dist)
end

local function isPlayingAnim(hum, animId)
    if not hum then return false end
    for _, track in ipairs(hum:GetPlayingAnimationTracks()) do
        local anim = track.Animation
        if anim then
            local id = anim.AnimationId
            if id and string.find(id, animId) then return true end
        end
    end
    return false
end

local function getNearestEnemy()
    local char = player.Character
    if not char then return nil, nil end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return nil, nil end
    local nearest = nil
    local nearestRoot = nil
    local nearestDist = math.huge
    local myPos = root.Position
    for _, other in ipairs(Players:GetPlayers()) do
        if other ~= player then
            local oChar = other.Character
            if oChar then
                local oRoot = oChar:FindFirstChild("HumanoidRootPart")
                if oRoot then
                    local dist = (myPos - oRoot.Position).Magnitude
                    if dist < nearestDist then
                        nearestDist = dist
                        nearest = oChar
                        nearestRoot = oRoot
                    end
                end
            end
        end
    end
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("Model") and obj ~= char then
            local hum = obj:FindFirstChild("Humanoid")
            local oRoot = obj:FindFirstChild("HumanoidRootPart")
            if hum and oRoot then
                local dist = (myPos - oRoot.Position).Magnitude
                if dist < nearestDist then
                    nearestDist = dist
                    nearest = obj
                    nearestRoot = oRoot
                end
            end
        end
    end
    if nearest and nearestRoot then
        return nearest, nearestRoot
    end
    return nil, nil
end

local function startTeleportLoop(enemyChar, enemyRoot)
    local startTime = tick()
    local myChar = player.Character
    if not myChar then return end
    local myRoot = myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot then return end
    local myHum = myChar:FindFirstChild("Humanoid")
    if not myHum then return end
    local enemyHum = enemyChar:FindFirstChild("Humanoid")
    if not enemyHum then return end
    myHum.PlatformStand = true
    local fixedOffset = randomOffset()
    local myPos = myRoot.Position
    local enemyPos = enemyRoot.Position
    local dirToEnemy = (enemyPos - myPos).Unit
    local facePos = enemyPos - dirToEnemy * SupaTech.FaceDistance
    facePos = Vector3.new(facePos.X, enemyPos.Y, facePos.Z)
    myRoot.CFrame = CFrame.new(facePos + fixedOffset) * CFrame.Angles(math.rad(SupaTech.UpwardAngle), 0, 0)
    task.wait(0.1)
    local footPos = enemyRoot.Position - Vector3.new(0, SupaTech.TeleportOffset, 0)
    myRoot.CFrame = CFrame.new(footPos + fixedOffset) * CFrame.Angles(math.rad(SupaTech.UpwardAngle), 0, 0)
    while tick() - startTime < SupaTech.TeleportDuration do
        if not SupaTech.Enabled then break end
        if not enemyRoot or not enemyRoot.Parent then break end
        if not myRoot or not myRoot.Parent then break end
        if not myHum then break end
        local currentEnemyHum = enemyChar:FindFirstChild("Humanoid")
        if not currentEnemyHum then break end
        if isPlayingAnim(currentEnemyHum, SupaTech.EndAnimId) or isPlayingAnim(currentEnemyHum, SupaTech.RagdollAnimId) then break end
        local newFootPos = enemyRoot.Position - Vector3.new(0, SupaTech.TeleportOffset, 0)
        myRoot.CFrame = CFrame.new(newFootPos + fixedOffset) * CFrame.Angles(math.rad(SupaTech.UpwardAngle), 0, 0)
        task.wait()
    end
    if enemyRoot and enemyRoot.Parent and myRoot and myRoot.Parent then
        local finalFootPos = enemyRoot.Position - Vector3.new(0, SupaTech.VerticalOffset, 0)
        myRoot.CFrame = CFrame.new(finalFootPos + fixedOffset) * CFrame.Angles(math.rad(SupaTech.UpwardAngle), 0, 0)
    end
    myHum.PlatformStand = false
end

local function executeSupaTech()
    if SupaTech.IsExecuting then return end
    if SupaTech.IsCooldown then return end
    SupaTech.IsExecuting = true
    local enemyChar, enemyRoot = getNearestEnemy()
    if not enemyChar or not enemyRoot then
        SupaTech.IsExecuting = false
        return
    end
    local enemyHumCheck = enemyChar:FindFirstChild("Humanoid")
    if enemyHumCheck and (isPlayingAnim(enemyHumCheck, SupaTech.RagdollAnimId) or isPlayingAnim(enemyHumCheck, SupaTech.EndAnimId)) then
        SupaTech.IsExecuting = false
        return
    end
    task.wait(SupaTech.WaitBeforeAction)
    if not enemyChar.Parent or not enemyRoot.Parent then
        SupaTech.IsExecuting = false
        return
    end
    local enemyHum2 = enemyChar:FindFirstChild("Humanoid")
    if enemyHum2 and (isPlayingAnim(enemyHum2, SupaTech.RagdollAnimId) or isPlayingAnim(enemyHum2, SupaTech.EndAnimId)) then
        SupaTech.IsExecuting = false
        return
    end
    doDash()
    startTeleportLoop(enemyChar, enemyRoot)
    SupaTech.IsExecuting = false
end

    end
end

-- ===== AntiFling (修复残留 & 恢复原始碰撞) =====
local AntiFling = {
    Enabled = false,
    MaxLinear = 200,
    MaxAngular = 100,
    SafeCFrame = nil,
    BodyMovers = {"BodyVelocity", "BodyAngularVelocity", "BodyForce", "BodyThrust", "BodyPosition", "BodyGyro"},
    Conns = {},
    HeartbeatCon = nil,
    OriginalCanCollide = {},
    RestoreTask = nil,
}

local function RecordOriginalCollide()
    local char = player.Character
    if not char then return end
    AntiFling.OriginalCanCollide = {}
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") then
            AntiFling.OriginalCanCollide[part] = part.CanCollide
        end
    end
end

local function RestoreOriginalCollide()
    if AntiFling.RestoreTask then
        task.cancel(AntiFling.RestoreTask)
        AntiFling.RestoreTask = nil
    end
    for part, original in pairs(AntiFling.OriginalCanCollide) do
        if part and part.Parent then
            part.CanCollide = original
            part.AssemblyLinearVelocity = Vector3.zero
            part.AssemblyAngularVelocity = Vector3.zero
            part.Velocity = Vector3.zero
            part.RotVelocity = Vector3.zero
        end
    end
    AntiFling.OriginalCanCollide = {}
    local char = player.Character
    if char then
        for _, part in ipairs(char:GetDescendants()) do
            if part:IsA("BasePart") then
                part.AssemblyLinearVelocity = Vector3.zero
                part.AssemblyAngularVelocity = Vector3.zero
                part.Velocity = Vector3.zero
                part.RotVelocity = Vector3.zero
            end
        end
    end
end

local function FlingNeutralize()
    if not AntiFling.Enabled then return end
    local char = player.Character
    if not char then return end
    for _, obj in ipairs(char:GetDescendants()) do
        if table.find(AntiFling.BodyMovers, obj.ClassName) then
            pcall(function() obj:Destroy() end)
        end
    end
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") then
            part.CanCollide = false
            part.AssemblyLinearVelocity = Vector3.zero
            part.AssemblyAngularVelocity = Vector3.zero
            part.Velocity = Vector3.zero
            part.RotVelocity = Vector3.zero
        end
    end
    local root = char:FindFirstChild("HumanoidRootPart")
    if root and AntiFling.SafeCFrame then
        root.CFrame = AntiFling.SafeCFrame
        char:SetPrimaryPartCFrame(AntiFling.SafeCFrame)
    end
    if AntiFling.RestoreTask then
        task.cancel(AntiFling.RestoreTask)
    end
    AntiFling.RestoreTask = task.delay(0.4, function()
        if char and char.Parent then
            for part, original in pairs(AntiFling.OriginalCanCollide) do
                if part and part.Parent then
                    part.CanCollide = original
                end
            end
        end
        AntiFling.RestoreTask = nil
    end)
    ShowNotification("检测到被甩飞，已 neutralize", Color3.fromRGB(255, 100, 0))
end

local function WatchBodyMovers(char)
    local con = char.DescendantAdded:Connect(function(child)
        if not AntiFling.Enabled then return end
        if table.find(AntiFling.BodyMovers, child.ClassName) then
            child:Destroy()
            ShowNotification("拦截异常 " .. child.ClassName, Color3.fromRGB(255, 100, 0))
            FlingNeutralize()
        end
    end)
    table.insert(AntiFling.Conns, con)
end

local function AntiFlingHeartbeat()
    if not AntiFling.Enabled then return end
    local char = player.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end
    local linear = root.AssemblyLinearVelocity.Magnitude
    local angular = root.AssemblyAngularVelocity.Magnitude
    if linear < 60 and angular < 40 then
        AntiFling.SafeCFrame = root.CFrame
    end
    if linear > AntiFling.MaxLinear or angular > AntiFling.MaxAngular then
        FlingNeutralize()
    end
end

        if AntiFling.Enabled then
            RecordOriginalCollide()
        end
    end
end

-- ===== 甩飞函数（来自皮脚本，增加距离检测与视角恢复） =====
FlingPlayer = function(TargetPlayer)
    local Character = player.Character
    local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")
    local RootPart = Humanoid and Humanoid.RootPart

    if not TargetPlayer or not TargetPlayer.Character then
        ShowNotification("目标无效", Color3.fromRGB(255, 50, 50))
        return
    end

    local TCharacter = TargetPlayer.Character
    local THumanoid
    local TRootPart
    local THead
    local Accessory
    local Handle

    if TCharacter:FindFirstChildOfClass("Humanoid") then
        THumanoid = TCharacter:FindFirstChildOfClass("Humanoid")
    end
    if THumanoid and THumanoid.RootPart then
        TRootPart = THumanoid.RootPart
    end
    if TCharacter:FindFirstChild("Head") then
        THead = TCharacter.Head
    end
    if TCharacter:FindFirstChildOfClass("Accessory") then
        Accessory = TCharacter:FindFirstChildOfClass("Accessory")
    end
    if Accessory and Accessory:FindFirstChild("Handle") then
        Handle = Accessory.Handle
    end

    if Character and Humanoid and RootPart then
        if RootPart.Velocity.Magnitude < 50 then
            local OldPos = RootPart.CFrame
            getgenv().OldPos = OldPos
        end
        if THumanoid and THumanoid.Sit then
            ShowNotification("目标正在坐着", Color3.fromRGB(255, 200, 0))
            return
        end

        -- 记录目标初始位置（用于距离检测）
        local targetStartPos = (TRootPart or THead or Handle).Position

        if THead then
            workspace.CurrentCamera.CameraSubject = THead
        elseif not THead and Handle then
            workspace.CurrentCamera.CameraSubject = Handle
        elseif THumanoid and TRootPart then
            workspace.CurrentCamera.CameraSubject = THumanoid
        end
        if not TCharacter:FindFirstChildWhichIsA("BasePart") then
            ShowNotification("目标缺少部件", Color3.fromRGB(255, 50, 50))
            return
        end

        local oldFPDH = workspace.FallenPartsDestroyHeight
        workspace.FallenPartsDestroyHeight = 0/0

        local FPos = function(BasePart, Pos, Ang)
            RootPart.CFrame = CFrame.new(BasePart.Position) * Pos * Ang
            Character:SetPrimaryPartCFrame(CFrame.new(BasePart.Position) * Pos * Ang)
            RootPart.Velocity = Vector3.new(9e7, 9e7 * 10, 9e7)
            RootPart.RotVelocity = Vector3.new(9e8, 9e8, 9e8)
        end

        local SFBasePart = function(BasePart)
            local TimeToWait = 2
            local Time = tick()
            local Angle = 0

            repeat
                -- ★ 距离检测：如果目标离开初始位置超过 100 格，立即恢复视角并跳出
                if (BasePart.Position - targetStartPos).Magnitude > 100 then
                    workspace.CurrentCamera.CameraSubject = Humanoid
                    ShowNotification("目标飞出太远，视角已恢复", Color3.fromRGB(255, 200, 100))
                    break
                end

                if RootPart and THumanoid then
                    if BasePart.Velocity.Magnitude < 50 then
                        Angle = Angle + 100

                        FPos(BasePart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle),0 ,0))
                        task.wait()

                        FPos(BasePart, CFrame.new(0, -1.5, 0) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                        task.wait()

                        FPos(BasePart, CFrame.new(2.25, 1.5, -2.25) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                        task.wait()

                        FPos(BasePart, CFrame.new(-2.25, -1.5, 2.25) + THumanoid.MoveDirection * BasePart.Velocity.Magnitude / 1.25, CFrame.Angles(math.rad(Angle), 0, 0))
                        task.wait()

                        FPos(BasePart, CFrame.new(0, 1.5, 0) + THumanoid.MoveDirection, CFrame.Angles(math.rad(Angle), 0, 0))
                        task.wait()

                        FPos(BasePart, CFrame.new(0, -1.5, 0) + THumanoid.MoveDirection, CFrame.Angles(math.rad(Angle), 0, 0))
                        task.wait()
                    else
                        FPos(BasePart, CFrame.new(0, 1.5, THumanoid.WalkSpeed), CFrame.Angles(math.rad(90), 0, 0))
                        task.wait()

                        FPos(BasePart, CFrame.new(0, -1.5, -THumanoid.WalkSpeed), CFrame.Angles(0, 0, 0))
                        task.wait()

                        FPos(BasePart, CFrame.new(0, 1.5, THumanoid.WalkSpeed), CFrame.Angles(math.rad(90), 0, 0))
                        task.wait()

                        FPos(BasePart, CFrame.new(0, 1.5, TRootPart.Velocity.Magnitude / 1.25), CFrame.Angles(math.rad(90), 0, 0))
                        task.wait()

                        FPos(BasePart, CFrame.new(0, -1.5, -TRootPart.Velocity.Magnitude / 1.25), CFrame.Angles(0, 0, 0))
                        task.wait()

                        FPos(BasePart, CFrame.new(0, 1.5, TRootPart.Velocity.Magnitude / 1.25), CFrame.Angles(math.rad(90), 0, 0))
                        task.wait()

                        FPos(BasePart, CFrame.new(0, -1.5, 0), CFrame.Angles(math.rad(90), 0, 0))
                        task.wait()

                        FPos(BasePart, CFrame.new(0, -1.5, 0), CFrame.Angles(0, 0, 0))
                        task.wait()

                        FPos(BasePart, CFrame.new(0, -1.5 ,0), CFrame.Angles(math.rad(-90), 0, 0))
                        task.wait()

                        FPos(BasePart, CFrame.new(0, -1.5, 0), CFrame.Angles(0, 0, 0))
                        task.wait()
                    end
                else
                    break
                end
            until BasePart.Velocity.Magnitude > 500 or BasePart.Parent ~= TargetPlayer.Character or TargetPlayer.Parent ~= Players or not TargetPlayer.Character == TCharacter or THumanoid.Sit or Humanoid.Health <= 0 or tick() > Time + TimeToWait

            -- 循环结束后，确保视角恢复到自己
            workspace.CurrentCamera.CameraSubject = Humanoid
        end

        local BV = Instance.new("BodyVelocity")
        BV.Name = "EpixVel"
        BV.Parent = RootPart
        BV.Velocity = Vector3.new(9e8, 9e8, 9e8)
        BV.MaxForce = Vector3.new(1/0, 1/0, 1/0)

        Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, false)

        if TRootPart and THead then
            if (TRootPart.CFrame.p - THead.CFrame.p).Magnitude > 5 then
                SFBasePart(THead)
            else
                SFBasePart(TRootPart)
            end
        elseif TRootPart and not THead then
            SFBasePart(TRootPart)
        elseif not TRootPart and THead then
            SFBasePart(THead)
        elseif not TRootPart and not THead and Accessory and Handle then
            SFBasePart(Handle)
        else
            ShowNotification("目标缺少可用部件", Color3.fromRGB(255, 50, 50))
            BV:Destroy()
            Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
            workspace.CurrentCamera.CameraSubject = Humanoid
            workspace.FallenPartsDestroyHeight = oldFPDH
            return
        end

        BV:Destroy()
        Humanoid:SetStateEnabled(Enum.HumanoidStateType.Seated, true)
        workspace.CurrentCamera.CameraSubject = Humanoid

        local oldPos = getgenv().OldPos or RootPart.CFrame
        repeat
            RootPart.CFrame = oldPos * CFrame.new(0, .5, 0)
            Character:SetPrimaryPartCFrame(oldPos * CFrame.new(0, .5, 0))
            Humanoid:ChangeState("GettingUp")
            for _, part in ipairs(Character:GetChildren()) do
                if part:IsA("BasePart") then
                    part.Velocity = Vector3.new()
                    part.RotVelocity = Vector3.new()
                end
            end
            task.wait()
        until (RootPart.Position - oldPos.p).Magnitude < 25

        RootPart.CFrame = oldPos
        Character:SetPrimaryPartCFrame(oldPos)
        for _, part in ipairs(Character:GetChildren()) do
            if part:IsA("BasePart") then
                part.Velocity = Vector3.new()
                part.RotVelocity = Vector3.new()
            end
        end
        workspace.FallenPartsDestroyHeight = oldFPDH
        ShowNotification("甩飞完成", Color3.fromRGB(100, 200, 255))
    else
        ShowNotification("角色或 Humanoid 丢失", Color3.fromRGB(255, 50, 50))
    end
end

local function Invis_RefreshParts()
    InvisData.char = player.Character or player.CharacterAdded:Wait()
    InvisData.hum = InvisData.char:WaitForChild('Humanoid')
    InvisData.root = InvisData.char:WaitForChild('HumanoidRootPart')
    InvisData.parts = {}
    for _, part in ipairs(InvisData.char:GetDescendants()) do
        if part:IsA('BasePart') and part.Transparency == 0 then
            table.insert(InvisData.parts, part)
        end
    end
end

local function Invis_Toggle()
    -- 与隐匿（假身）互斥：两者都压部件透明度、抢相机
    if not InvisData.enabled and Rev.Invis.Enabled and Rev.StopInvis then
        Rev.StopInvis(true)
    end
    InvisData.enabled = not InvisData.enabled
    for _, part in ipairs(InvisData.parts) do
        if part and part.Parent then
            part.Transparency = InvisData.enabled and 0.5 or 0
        end
    end
    UpdateToggleButton(InvisButton, InvisData.enabled, "隐身:开启", "隐身:关闭")
    ShowNotification(InvisData.enabled and "隐身已开启" or "隐身已关闭", InvisData.enabled and Color3.fromRGB(0, 255, 100) or Color3.fromRGB(255, 100, 0))
end

end

        if hum then

            ShowNotification("已自杀，即将重生", Color3.fromRGB(255, 200, 0))
        else
            ShowNotification("未找到 Humanoid", Color3.fromRGB(255, 50, 50))
        end
    else
        ShowNotification("角色不存在", Color3.fromRGB(255, 50, 50))
    end
end)

            end

        end

end)

RunService.Heartbeat:Connect(function()
    if not SupaTech.Enabled then
        if SupaTech.IsExecuting then SupaTech.IsExecuting = false end
        return
    end
    if SupaTech.IsExecuting then return end
    if SupaTech.IsCooldown then return end
    local char = player.Character
    if not char then return end
    local hum = char:FindFirstChild("Humanoid")
    if not hum then return end
    if isPlayingAnim(hum, SupaTech.AnimId) then
        local enemyChar, enemyRoot = getNearestEnemy()
        if enemyChar and enemyRoot then
            local myRoot = char:FindFirstChild("HumanoidRootPart")
            if myRoot then
                local dist = (myRoot.Position - enemyRoot.Position).Magnitude
                if dist <= SupaTech.NearbyRange then
                    local enemyHumCheck = enemyChar:FindFirstChild("Humanoid")
                    if not (enemyHumCheck and (isPlayingAnim(enemyHumCheck, SupaTech.RagdollAnimId) or isPlayingAnim(enemyHumCheck, SupaTech.EndAnimId))) then
                        task.spawn(executeSupaTech)
                    end
                end
            end
        end
    end
end)
